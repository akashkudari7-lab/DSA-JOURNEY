## 1. Basic Pointer & Double Pointer Code

```cpp
#include <iostream>
#include <vector>
using namespace std;

int main() {
    int a = 10;
    int* ptr = &a;
    int** parptr = &ptr;    

    cout << ptr << endl;
    cout << &a << endl;
    cout << *(&a) << endl;
    cout << *ptr << endl;
    cout << **parptr << endl;
    cout << ptr << endl;
    cout << *parptr << endl;
    return 0;
}
```

## 2. Array Pointers (Pointer Arithmetic & Distance)

Pointers can be subtracted to find the distance between elements, and can also be compared:

```cpp
#include <iostream>
#include <vector>
using namespace std;

int main() {
    int arr[] = {10, 20, 30, 40, 50};
    int* ptr2 = arr;
    int* ptr1 = ptr2 + 2;

    cout << ptr1 - ptr2 << endl; // Prints distance in elements (2)
    return 0;
}
```
